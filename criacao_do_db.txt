-- Criar tabela PESQUISADOR
CREATE TABLE PESQUISADOR (
    CPF TEXT PRIMARY KEY,
    NOME TEXT NOT NULL,
    INSTITUICAO TEXT NOT NULL,
    NASCIMENTO TEXT NOT NULL
);

-- Criar tabela EVENTO
CREATE TABLE EVENTO (
    COD INTEGER PRIMARY KEY,
    NOME TEXT NOT NULL,
    SIGLA TEXT NOT NULL,
    ANO INTEGER NOT NULL
);

-- Criar tabela ARTIGO
CREATE TABLE ARTIGO (
    MAT INTEGER PRIMARY KEY,
    TITULO TEXT NOT NULL,
    NOTA REAL NOT NULL,
    IDIOMA TEXT NOT NULL,
    COD INTEGER,
    FOREIGN KEY (COD) REFERENCES EVENTO(COD)
);

-- Criar tabela ESCREVE (relacionamento muitos-para-muitos entre PESQUISADOR e ARTIGO)
CREATE TABLE ESCREVE (
    CPF TEXT,
    MAT INTEGER,
    PRIMARY KEY (CPF, MAT),
    FOREIGN KEY (CPF) REFERENCES PESQUISADOR(CPF),
    FOREIGN KEY (MAT) REFERENCES ARTIGO(MAT)
);
